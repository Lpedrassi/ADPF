rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // TRAVA TEMPORÁRIA (02/11/2026, até reestruturar a segurança): quem criou login e ainda é "convidado"
    // não lê nem grava nada do portal. Para liberar, o admin muda o perfil no Gerenciar Perfis.
    function aprovado() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/usuarios/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.perfil != "convidado";
    }

    function isAdmin() {
      return aprovado() &&
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.perfil == "admin";
    }

    function isAdministrativo() {
      return aprovado() &&
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.perfil == "administrativo";
    }

    function podeEscalarNaSala(salaId) {
      let sala = get(/databases/$(database)/documents/salas/$(salaId)).data;
      return sala.professorUid == request.auth.uid ||
        (sala.professoresUids is list && sala.professoresUids.hasAny([request.auth.uid]));
    }

    // Usuários
    match /usuarios/{userId} {
      allow get: if request.auth != null && (request.auth.uid == userId || isAdmin());
      allow list: if request.auth != null && (
        isAdmin() ||
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.perfil == "professor" ||
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.perfil == "administrativo"
      );
      // Cadastro pela própria pessoa: sempre como convidado (ninguém cria a própria conta como admin).
      // A equipe cria alunos pendentes de verificação.
      allow create: if request.auth != null && (
        (request.auth.uid == userId && request.resource.data.perfil == "convidado") ||
        // Merge pelo código de acesso: a conta nova assume o cadastro da Secretaria (contaAnterior),
        // desde que o código exista, não tenha sido usado e seja desse cadastro
        (request.auth.uid == userId && request.resource.data.perfil == "aluno"
          && request.resource.data.cadastroPendente == false
          && request.resource.data.codigoVinculo is string
          && get(/databases/$(database)/documents/codigosAcesso/$(request.resource.data.codigoVinculo)).data.usado == false
          && get(/databases/$(database)/documents/codigosAcesso/$(request.resource.data.codigoVinculo)).data.usuarioId == request.resource.data.contaAnterior) ||
        ((isAdmin() || isAdministrativo() ||
          get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.perfil == "professor")
          && request.resource.data.cadastroPendente == true
          && request.resource.data.perfil == "aluno")
      );
      // Atualização: o admin altera tudo. A própria pessoa altera os seus dados, mas não perfil nem cadastroPendente,
      // e pode desativar a própria conta (ativo: false), nunca reativá-la.
      allow update: if request.auth != null && (
        isAdmin() ||
        (request.auth.uid == userId &&
         !request.resource.data.diff(resource.data).affectedKeys().hasAny(['perfil', 'cadastroPendente']) &&
         (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['ativo']) || request.resource.data.ativo == false))
      );
      // Exclusão: admin, ou a conta nova apagando o cadastro antigo que ela assumiu (merge)
      allow delete: if isAdmin() || (request.auth != null
        && get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.contaAnterior == userId);
    }

    // Índice do celular (hash -> { uid, email }), usado só para validar duplicidade no Perfil.
    // Leitura só para quem está logado; gravação só pelo dono da conta (ou admin).
    match /loginCelular/{hash} {
      allow get: if request.auth != null;
      allow create: if request.auth != null
                    && (request.resource.data.uid == request.auth.uid || isAdmin());
      allow update, delete: if request.auth != null
                    && (resource.data.uid == request.auth.uid || isAdmin());
    }

    // Índice de nome de usuário (nome -> { uid, email }), usado só para validar duplicidade no Perfil.
    // Leitura só para quem está logado; gravação só pelo dono (ou admin).
    match /loginUsuario/{nome} {
      allow get: if request.auth != null;
      allow create: if request.auth != null
                    && (request.resource.data.uid == request.auth.uid || isAdmin());
      allow update: if request.auth != null
                    && (resource.data.uid == request.auth.uid || isAdmin());
      allow delete: if isAdmin();
    }

    match /agenda/{eventoId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /agenda_meta/{docId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /salas/{salaId} {
      allow read: if aprovado();
      allow write: if isAdmin();
    }
    match /relatoriosAula/{relatorioId} {
      allow read: if aprovado();
      allow create, update: if aprovado() && (
        isAdmin() ||
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.perfil == "administrativo" ||
        podeEscalarNaSala(request.resource.data.salaId)
      );
      allow delete: if aprovado() && (isAdmin() || isAdministrativo());
    }
    // Códigos de acesso (Secretaria/admin geram). O aluno confere um código pelo próprio código
    // (get, nunca list) e o marca como usado no merge.
    match /codigosAcesso/{codigo} {
      allow get: if request.auth != null;
      allow create: if request.auth != null && (isAdmin() || isAdministrativo())
                    && request.resource.data.usado == false;
      allow update: if request.auth != null && (
        isAdmin() || isAdministrativo() ||
        (resource.data.usado == false && request.resource.data.usado == true
          && request.resource.data.usadoPorUid == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['usado', 'usadoPorUid', 'usadoEm']))
      );
      allow delete: if isAdmin();
    }
    match /matriculas/{matriculaId} {
      allow read: if aprovado();
      allow create, update: if aprovado() && (
        isAdmin() ||
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.perfil == "administrativo" ||
        podeEscalarNaSala(request.resource.data.salaId) ||
        // Merge: a conta nova move a matrícula da conta antiga (contaAnterior) para si
        (resource.data.uid == get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.contaAnterior
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['uid']))
      );
    }
    match /licoes/{licaoId} {
      allow read: if aprovado();
      allow write: if isAdmin();
    }
    match /arquivosSala/{arquivoId} {
      allow read: if aprovado();
      allow create, update: if aprovado() && (
        isAdmin() ||
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.perfil == "administrativo" ||
        podeEscalarNaSala(request.resource.data.salaId)
      );
    }
    match /escalaEbd/{escalaId} {
      allow read: if aprovado();
      allow write: if isAdmin();
    }
    match /caixaMovimentacoes/{movimentacaoId} {
      allow read, write: if aprovado() && (isAdmin() || isAdministrativo());
    }
    // Agendas da Secretaria (compromissos): só admin e administrativo
    match /compromissosSecretaria/{compromissoId} {
      allow read, write: if aprovado() && (isAdmin() || isAdministrativo());
    }
    // Avisos para usuários (convites de compromisso): admin e administrativo criam;
    // o destinatário lê o próprio aviso e só pode marcá-lo como lido
    match /avisosUsuarios/{avisoId} {
      allow read: if aprovado() && resource.data.uidDestino == request.auth.uid;
      allow create: if aprovado() && (isAdmin() || isAdministrativo());
      allow update: if aprovado() && resource.data.uidDestino == request.auth.uid
                    && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['lido']);
      allow delete: if aprovado() && (isAdmin() || isAdministrativo());
    }
    // Perfis de acesso (Gerenciar Perfis): todos os logados leem (para montar o menu);
    // só o Administrador Master cria, edita e exclui
    match /perfisAcesso/{perfilId} {
      allow read: if aprovado();
      allow write: if isAdmin();
    }
    // Categorias dos posts do carrosel (Secretaria/Administração editam)
    match /configPosts/{docId} {
      allow read: if aprovado() && (isAdmin() || isAdministrativo());
      allow write: if aprovado() && (isAdmin() || isAdministrativo());
    }
    match /materiaisEstoque/{materialId} {
      allow read, write: if aprovado() && (isAdmin() || isAdministrativo());
    }
    match /materiaisMovimentacoes/{movimentacaoId} {
      allow read, write: if aprovado() && (isAdmin() || isAdministrativo());
    }
    // Pedidos de revista: o aluno cria só o próprio pedido e lê o próprio;
    // a Secretaria lê todos e corrige (pagamento, status)
    match /pedidosRevistas/{pedidoId} {
      allow create: if aprovado() && (
        request.resource.data.uid == request.auth.uid ||
        isAdmin() || isAdministrativo()
      );
      allow read: if aprovado() && (
        (resource != null && resource.data.uid == request.auth.uid) ||
        isAdmin() || isAdministrativo()
      );
      // Aluno só pede ou desfaz cancelamento no próprio pedido, e só enquanto a Secretaria
      // estiver em "Solicitado pelo Aluno" (status pendente). Em "Pedido" ou "Entregue", fala com a secretaria.
      allow update: if aprovado() && (
        isAdmin() || isAdministrativo() ||
        (resource.data.uid == request.auth.uid &&
          !(resource.data.status in ['pedido', 'entregue']) &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['cancelamentoSolicitadoEm'])) ||
        // Merge: o pedido da conta antiga passa para a conta nova
        (resource.data.uid == get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.contaAnterior
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['uid']))
      );
      allow delete: if aprovado() && (isAdmin() || isAdministrativo());
    }
    // Valores das revistas (Secretaria cadastra; alunos leem para ver o total do pedido)
    // Listas compartilhadas do site (ex.: motivos de ausência). Todos leem; só o Administrador grava.
    match /configListas/{docId} {
      allow read: if aprovado();
      allow write: if isAdmin();
    }
    match /configRevistas/{docId} {
      allow read: if aprovado();
      allow write: if aprovado() && (isAdmin() || isAdministrativo());
    }

    // ===== NOVO: Turmas EBD (Escalas de Professores) =====
    match /turmas/{turmaId} {
      allow read: if true; // Public read for viewing EBD classes
      allow write: if isAdmin();

      // 13 lições por turma
      match /licoes/{licaoId} {
        allow read: if true;
        allow write: if isAdmin();
      }

      // 12 escalas mensais por turma
      match /escalas/{escalaId} {
        allow read: if true;
        allow write: if isAdmin();
      }
    }

    // ===== Posts/Notícias do Carousel =====
    match /noticias/{noticiaId} {
      allow read: if true; // Public read for carousel posts
      allow create, update, delete: if isAdmin();
    }
  }
}
